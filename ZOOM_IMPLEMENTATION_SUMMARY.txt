LIVE STATUS DETECTION (Automatic):
  ✓ Checks if today is Sunday
  ✓ Checks if current time >= 8:00 PM
  ✓ Checks if current time < 10:30 PM
  ✓ Returns LIVE or OFFLINE status

RESPONSE WHEN LIVE (Sunday 8:00-10:30 PM):
  {
    "status": "live",
    "data": {
      "title": "SUNDAY NIGHT PRAYER MEETING",
      "platform": "zoom",
      "meeting_url": "https://us06web.zoom.us/j/...",
      "start_time": "2025-12-21 20:00:00"
    }
  }

RESPONSE WHEN OFFLINE (Other times):
  {
    "status": "offline",
    "message": "Zoom service holds every Sunday by 8:00 PM"
  }
